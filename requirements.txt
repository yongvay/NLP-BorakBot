# Versions are the ones that produced the live demo, so they are pinned exactly.

#     pip install -r requirements.txt

# ---------------------------------------------------------------------------
# READ THIS FIRST ON WINDOWS — enable long paths, or the install WILL fail
# ---------------------------------------------------------------------------
# torch ships its third-party licences nested about seven directories deep. Under
# the default 260-character MAX_PATH that overflows, and pip dies partway with:
#
#     OSError: [WinError 206] The filename or extension is too long
#
# The damage is silent and confusing: torch/ unpacks, torchgen/ never does, and
# every later run fails with "ModuleNotFoundError: No module named 'torchgen'",
# which says nothing about path length. Do not chase torchgen — there is a
# package by that name on PyPI and it is NOT PyTorch's. Installing it hides the
# real fault.
#
# In an ADMINISTRATOR PowerShell, once per machine, then REBOOT:
#
#     New-ItemProperty -Path "HKLM:\SYSTEM\CurrentControlSet\Control\FileSystem" `
#       -Name LongPathsEnabled -Value 1 -PropertyType DWORD -Force
#
# No admin rights? Put the venv somewhere short instead — it does not have to
# live in the repo, whose own path already costs 65 characters:
#
#     py -3.13 -m venv C:\v\borak
#     C:\v\borak\Scripts\Activate.ps1
#
# ---------------------------------------------------------------------------
# ffmpeg is NOT a pip package and is NOT installed by this file
# ---------------------------------------------------------------------------
# Whisper shells out to it to decode and resample audio. Without it, every
# transcription raises FileNotFoundError.
#
#     winget install Gyan.FFmpeg      # then reopen the terminal
#     ffmpeg -version                 # must print a version
# ---------------------------------------------------------------------------

# CPU-only torch. The plain PyPI wheel bundles CUDA and is ~2.5 GB; this one is
# ~120 MB and is ample for whisper-small at demo scale (~7 s per utterance).
# The +cpu build exists only on PyTorch's own index, hence the extra index below.
# On a machine with an NVIDIA GPU, drop both lines and `pip install torch` —
# app/stt.py calls torch.cuda.is_available() and adapts on its own.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.13.0+cpu

# The reference Whisper implementation (Radford et al., 2023) — the one the
# proposal cites. Used only as a language detector; see app/stt.py.
openai-whisper==20250625

# Hosts mesolitica/malaysian-whisper-small-v3, which does all transcription.
# NOTE: this is transformers 5.x. app/stt.py depends on tokenizer.get_prompt_ids()
# and on how prompt_ids echo into the decoded output — _strip_prompt() exists for
# exactly that, and its behaviour is version-dependent. If a Colab or Kaggle run
# resolves a different major version, expect the transcripts to differ from the
# ones behind the reported 0.349 WER.
transformers==5.15.0

# The UI. st.audio_input (in-browser recording) needs >= 1.35.
streamlit==1.61.1
